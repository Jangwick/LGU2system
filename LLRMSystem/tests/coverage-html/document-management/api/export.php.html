412function formatFileSize($bytes) {

414        return number_format($bytes / 1073741824, 2) . ' GB';
415    } elseif ($bytes >= 1048576) {
416        return number_format($bytes / 1048576, 2) . ' MB';
417    } elseif ($bytes >= 1024) {
